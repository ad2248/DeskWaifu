cmake_minimum_required(VERSION 3.16)
project(ocr_communication_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
enable_testing()
add_executable(ocr-test-helper OcrTestHelper.cpp)
target_link_libraries(ocr-test-helper PRIVATE Qt6::Core)
add_executable(ocr-client-test OcrClientTest.cpp ../../../src/processor/ocr/OcrClient.cpp ../../../src/processor/ocr/OcrClient.hpp)
set_target_properties(ocr-client-test PROPERTIES AUTOMOC ON)
target_include_directories(ocr-client-test PRIVATE ../../../src/processor/ocr)
target_compile_definitions(ocr-client-test PRIVATE DESKWAIFU_OCR_HELPER_PATH="/nonexistent/deskwaifu-ocr")
target_link_libraries(ocr-client-test PRIVATE Qt6::Core)
add_test(NAME ocr-client COMMAND ocr-client-test $<TARGET_FILE:ocr-test-helper>)
set_tests_properties(ocr-client PROPERTIES TIMEOUT 30)
