cmake_minimum_required(VERSION 3.16)
project(cubism_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_UTILS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GLEW_X11 OFF CACHE BOOL "" FORCE)
set(GLEW_EGL ON CACHE BOOL "" FORCE)
add_subdirectory("${GLEW_ROOT}/build/cmake" glew EXCLUDE_FROM_ALL)
set(FRAMEWORK_SOURCE OpenGL)
add_compile_definitions(CSM_TARGET_LINUX_GL GLEW_STATIC GLEW_EGL)
include_directories("${GLEW_ROOT}/include" "${CUBISM_SDK_ROOT}/Core/include")
add_subdirectory("${CUBISM_SDK_ROOT}/Framework" framework)
add_library(cubism_bridge STATIC bridge.cpp)
target_link_libraries(cubism_bridge PRIVATE Framework glew_s GL EGL)
install(TARGETS cubism_bridge Framework glew_s ARCHIVE DESTINATION lib)
