  foreach(script smoke_isolation test_environment test_gate aur_packaging aur_binary_packaging release_packaging installed_testing mutation_configuration documentation_links)

    add_test(NAME ${test_name} COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/test_${script}.py)
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 30)

  endforeach()
