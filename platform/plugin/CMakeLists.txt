cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(deskwaifu_plugin_hosts LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET deskwaifu_plugin_sdk)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../sdk ${CMAKE_CURRENT_BINARY_DIR}/sdk)
endif()

add_library(deskwaifu_plugin_host STATIC PluginManager.cpp PluginCommand.cpp)
target_include_directories(deskwaifu_plugin_host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_link_libraries(deskwaifu_plugin_host PUBLIC deskwaifu_plugin_sdk)
target_compile_features(deskwaifu_plugin_host PUBLIC cxx_std_14)

if(DESKWAIFU_BUILD_GUI_PLUGIN_SDK)
  add_library(deskwaifu_gui_plugin_host STATIC GuiPluginManager.cpp)
  target_include_directories(deskwaifu_gui_plugin_host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
  target_link_libraries(deskwaifu_gui_plugin_host PUBLIC deskwaifu_gui_plugin_sdk)
  target_compile_features(deskwaifu_gui_plugin_host PUBLIC cxx_std_14)
endif()

if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
