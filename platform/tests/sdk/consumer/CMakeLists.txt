cmake_minimum_required(VERSION 3.16)
project(deskwaifu_installed_sdk_consumer LANGUAGES CXX)
include(CTest)
if(TEST_GUI)
  find_package(DeskWaifuPluginSDK REQUIRED COMPONENTS Gui)
else()
  find_package(DeskWaifuPluginSDK REQUIRED)
endif()
add_executable(installed-sdk-consumer ../PluginContractTest.cpp)
target_link_libraries(installed-sdk-consumer PRIVATE DeskWaifu::plugin_sdk)
target_compile_features(installed-sdk-consumer PRIVATE cxx_std_14)
if(TEST_GUI)
  target_link_libraries(installed-sdk-consumer PRIVATE DeskWaifu::gui_plugin_sdk)
  target_compile_definitions(installed-sdk-consumer PRIVATE DESKWAIFU_TEST_GUI)
endif()
add_test(NAME installed-sdk-contracts COMMAND installed-sdk-consumer)
