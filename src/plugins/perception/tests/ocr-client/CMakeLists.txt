cmake_minimum_required(VERSION 3.25)
project(ocr_communication_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(PRIVATE_ROOT "${ROOT}/src/plugins/perception/private")
add_executable(ocr-test-helper ${ROOT}/src/plugins/perception/tests/OcrTestHelper.cpp)
target_link_libraries(ocr-test-helper PRIVATE Qt6::Core)
add_executable(ocr-client-test ${ROOT}/src/plugins/perception/tests/OcrClientTest.cpp
  ${PRIVATE_ROOT}/processor/ocr/OcrClient.cpp ${PRIVATE_ROOT}/processor/ocr/OcrClient.hpp)
set_target_properties(ocr-client-test PROPERTIES AUTOMOC ON)
target_include_directories(ocr-client-test PRIVATE ${ROOT}/sdk ${PRIVATE_ROOT})
target_compile_definitions(ocr-client-test PRIVATE DESKWAIFU_OCR_HELPER_PATH="/nonexistent/deskwaifu-ocr")
target_link_libraries(ocr-client-test PRIVATE Qt6::Core)
add_test(NAME ocr-client COMMAND ocr-client-test $<TARGET_FILE:ocr-test-helper>)
set_tests_properties(ocr-client PROPERTIES TIMEOUT 30)
